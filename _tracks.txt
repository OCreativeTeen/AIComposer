187|        self,
188|        src_mp4: str,
189|        volume: float,
190|        *,
191|        start: float = 0.0,
192|        end: float | None = None,
193|        speed: float = 1.0,
194|    ):
195|        """按区间裁剪、变速。音量增益不是 1 时才改音量。"""
196|        wf = self.workflow
197|        if not wf:
198|            raise RuntimeError("工作流未就绪")
199|        ap = wf.ffmpeg_audio_processor
200|        fp = wf.ffmpeg_processor
201|        dur = float(fp.get_duration(src_mp4) or 0.0)
202|        st = max(0.0, float(start or 0.0))
203|        en = float(end if end is not None else dur)
204|        if en <= 0 or en > dur:
205|            en = dur
206|        spd = round(float(speed or 1.0), 3)
207|        vol = float(volume or 1.0)
208|
209|        mp4_out = fp.trim_video(src_mp4, st, en, volume=1.0, speed=spd)
210|        if not mp4_out or not os.path.isfile(mp4_out):
211|            raise RuntimeError("裁剪/变速失败")
212|
213|        raw = ap.extract_audio_from_video(mp4_out)
214|        if abs(vol - 1.0) < 0.001:
215|            if raw:
216|                return mp4_out, raw
217|            seg_len = fp.get_duration(mp4_out) or 0.0
218|            wav_out = ap.make_silence(seg_len) if seg_len > 0 else None
219|            if not wav_out:
220|                raise RuntimeError("无法生成临时 wav")
221|            return mp4_out, wav_out
222|
223|        if raw:
224|            raw_len = ap.get_duration(raw)
225|            if raw_len is None:
226|                raw_len = fp.get_duration(mp4_out) or 0.0
227|            wav_out = ap.audio_cut_fade(raw, 0, raw_len, 0, 0, volume=vol)
228|        else:
229|            seg_len = fp.get_duration(mp4_out) or 0.0
230|            wav_out = ap.make_silence(seg_len) if seg_len > 0 else None
231|        if not wav_out:
232|            raise RuntimeError("无法生成增益后的临时 wav")
233|        muxed = fp.add_audio_to_video(mp4_out, wav_out)
234|        if not muxed or not os.path.isfile(muxed):
235|            raise RuntimeError("无法生成增益后的临时 mp4")
236|        return muxed, wav_out
